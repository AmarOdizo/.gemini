<script>
        // Simple logic to toggle the unavailable state appearance
        document.querySelectorAll('input[type="checkbox"][id^="toggle_"]').forEach(checkbox => {
            checkbox.addEventListener('change', (e) => {
                const dayRow = e.target.closest('.border-b');
                const slotsContainer = dayRow.querySelector('.flex-1.space-y-3');
                
                if (e.target.checked) {
                    dayRow.classList.remove('bg-surface-container-low/30', 'opacity-75');
                    if (slotsContainer) slotsContainer.style.display = 'block';
                    // In a real app, you'd replace the "Unavailable" text with actual slots
                } else {
                    dayRow.classList.add('bg-surface-container-low/30', 'opacity-75');
                    if (slotsContainer) slotsContainer.style.display = 'none';
                    // Need to inject unavailable text if slots are hidden
                }
            });
        });
        
        // Hide header on scroll down for mobile
        let lastScrollY = window.scrollY;
        const mobileHeader = document.getElementById('mobile-header');
        
        window.addEventListener('scroll', () => {
            if (window.innerWidth < 768) {
                if (lastScrollY < window.scrollY) {
                    mobileHeader.style.transform = 'translateY(-100%)';
                } else {
                    mobileHeader.style.transform = 'translateY(0)';
                }
            }
            lastScrollY = window.scrollY;
        });
    </script>
