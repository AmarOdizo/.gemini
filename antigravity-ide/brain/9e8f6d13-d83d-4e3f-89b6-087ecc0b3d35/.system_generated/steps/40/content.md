  positions[i * 3 + 1] = (Math.random() - 0.2) * 6;
  positions[i * 3 + 2] = (Math.random() - 0.5) * 8;

// --- 7. INTERACTION CONTROLS (360 Orbit, Drag, Zoom, Hover) ---
let isDragging = false;
let previousMousePosition = { x: 0, y: 0 };
let targetRotationY = 0;
let targetRotationX = 0;
let currentRotationY = 0;
let currentRotationX = 0;
let targetZoom = 6.5;
let currentZoom = 6.5;

function onMouseDown(e) {
  isDragging = true;
  previousMousePosition = { x: e.clientX, y: e.clientY };
}

function onMouseMove(e) {
  if (isDragging) {
    const deltaX = e.clientX - previousMousePosition.x;
    const deltaY = e.clientY - previousMousePosition.y;

    targetRotationY += deltaX * 0.007;
    targetRotationX = Math.max(-0.45, Math.min(0.45, targetRotationX + deltaY * 0.005));

    previousMousePosition = { x: e.clientX, y: e.clientY };
  }
}

function onMouseUp() {
  isDragging = false;
}

function onWheel(e) {
  targetZoom += e.deltaY * 0.003;
  targetZoom = Math.max(4.0, Math.min(9.5, targetZoom));
}

// Touch controls for mobile/tablet 360 rotation
function onTouchStart(e) {
  if (e.touches.length === 1) {
    isDragging = true;
    previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY };
  }
}

function onTouchMove(e) {
  if (isDragging && e.touches.length === 1) {
    const deltaX = e.touches[0].clientX - previousMousePosition.x;
    const deltaY = e.touches[0].clientY - previousMousePosition.y;
    targetRotationY += deltaX * 0.008;
    targetRotationX = Math.max(-0.4, Math.min(0.4, targetRotationX + deltaY * 0.006));
    previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY };
  }
}

function onTouchEnd() {
  isDragging = false;
}

window.addEventListener('mousemove', onMouseMove);
window.addEventListener('mouseup', onMouseUp);
window.addEventListener('wheel', onWheel, { passive: true });
window.addEventListener('touchstart', onTouchStart, { passive: true });
window.addEventListener('touchmove', onTouchMove, { passive: true });
window.addEventListener('touchend', onTouchEnd);

// Interactive Click trigger: Raycaster to bounce dog or cat when clicked
const raycaster = new THREE.Raycaster();
const mouseVec = new THREE.Vector2();

window.addEventListener('click', (e) => {
  const rect = renderer.domElement.getBoundingClientRect();
  mouseVec.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
  mouseVec.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
  raycaster.setFromCamera(mouseVec, camera);

  const intersects = raycaster.intersectObjects([dogHead, catHead, stethObj, heartGroup], true);
  if (intersects.length > 0) {

    dogGroup.position.y = -0.3;
    catGroup.position.y = -0.4;
    setTimeout(() => {
      dogGroup.position.y = -0.6;
      catGroup.position.y = -0.65;
    }, 250);
  }
});

// Window resize listener
function handleResize() {
  const w = container.clientWidth || window.innerWidth;
  const h = container.clientHeight || window.innerHeight;
  camera.aspect = w / h;
  camera.updateProjectionMatrix();
  renderer.setSize(w, h);
}
window.addEventListener('resize', handleResize);

// Animation Loop
let clock = new THREE.Clock();

function animate() {
  requestAnimationFrame(animate);
  const elapsed = clock.getElapsedTime();

  // Smooth rotation damping
  currentRotationY += (targetRotationY - currentRotationY) * 0.08;
  currentRotationX += (targetRotationX - currentRotationX) * 0.08;
  worldGroup.rotation.y = currentRotationY + (isDragging ? 0 : Math.sin(elapsed * 0.3) * 0.15);
  worldGroup.rotation.x = currentRotationX;

  // Smooth zoom damping
  currentZoom += (targetZoom - currentZoom) * 0.08;
  camera.position.z = currentZoom;

  // Gentle character breathing & idling animations
  dogHeadGroup.rotation.z = Math.sin(elapsed * 2.0) * 0.06;
  dogHeadGroup.rotation.y = Math.cos(elapsed * 1.5) * 0.08;
  tail.rotation.z = Math.sin(elapsed * 8.0) * 0.35; // Wagging tail

  catHeadGroup.rotation.x = Math.sin(elapsed * 1.2) * 0.05;
  catTail.rotation.z = 0.5 + Math.sin(elapsed * 2.5) * 0.18;

  // Float items
  floatingItems.forEach(item => {
    item.mesh.position.y = item.basePos.y + Math.sin(elapsed * item.speed) * item.amp;
    item.mesh.rotation.y += item.rotSpeed;
    item.mesh.rotation.x += item.rotSpeed * 0.5;
  });

  // Rotate particles gently
  particles.rotation.y = elapsed * 0.03;

  renderer.render(scene, camera);
}

animate();

})();
