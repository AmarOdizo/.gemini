<script>
        document.addEventListener('DOMContentLoaded', () => {
            const form = document.getElementById('registrationForm');
            const fullNameInput = document.getElementById('fullName');
            const emailInput = document.getElementById('email');
            const passwordInput = document.getElementById('password');
            const confirmPasswordInput = document.getElementById('confirmPassword');
            const termsInput = document.getElementById('terms');

            const togglePasswordBtns = document.querySelectorAll('.toggle-password');

            // Password visibility toggle
            togglePasswordBtns.forEach(btn => {
                btn.addEventListener('click', function() {
                    const input = this.previousElementSibling;
                    const icon = this.querySelector('.material-symbols-outlined');
                    if (input.type === 'password') {
                        input.type = 'text';
                        icon.textContent = 'visibility_off';
                    } else {
                        input.type = 'password';
                        icon.textContent = 'visibility';
                    }
                });
            });

            // Validation Helper
            function showError(inputElement, errorElementId, show) {
                const errorElement = document.getElementById(errorElementId);
                if (show) {
                    inputElement.classList.add('border-error', 'focus:ring-error');
                    inputElement.classList.remove('border-outline-variant', 'focus:ring-primary');
                    errorElement.classList.remove('hidden');
                } else {
                    inputElement.classList.remove('border-error', 'focus:ring-error');
                    inputElement.classList.add('border-outline-variant', 'focus:ring-primary');
                    errorElement.classList.add('hidden');
                }
            }

            // Real-time validation
            emailInput.addEventListener('input', () => {
                 const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
                 showError(emailInput, 'emailError', !emailRegex.test(emailInput.value) && emailInput.value !== '');
            });

            passwordInput.addEventListener('input', () => {
                showError(passwordInput, 'passwordError', passwordInput.value.length > 0 && passwordInput.value.length < 8);
                if(confirmPasswordInput.value) {
                     showError(confirmPasswordInput, 'confirmPasswordError', passwordInput.value !== confirmPasswordInput.value);
                }
            });

            confirmPasswordInput.addEventListener('input', () => {
                showError(confirmPasswordInput, 'confirmPasswordError', passwordInput.value !== confirmPasswordInput.value && confirmPasswordInput.value !== '');
            });

            // Form Submit Validation
            form.addEventListener('submit', (e) => {
                e.preventDefault();
                let isValid = true;

                // Full Name
                if (!fullNameInput.value.trim()) {
                    showError(fullNameInput, 'fullNameError', true);
                    isValid = false;
                } else {
                    showError(fullNameInput, 'fullNameError', false);
                }

                // Email
                const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
                if (!emailRegex.test(emailInput.value)) {
                    showError(emailInput, 'emailError', true);
                    isValid = false;
                } else {
                    showError(emailInput, 'emailError', false);
                }

                // Password
                if (passwordInput.value.length < 8) {
                    showError(passwordInput, 'passwordError', true);
                    isValid = false;
                } else {
                    showError(passwordInput, 'passwordError', false);
                }

                // Confirm Password
                if (passwordInput.value !== confirmPasswordInput.value || !confirmPasswordInput.value) {
                    showError(confirmPasswordInput, 'confirmPasswordError', true);
                    isValid = false;
                } else {
                     showError(confirmPasswordInput, 'confirmPasswordError', false);
                }

                // Terms
                const termsError = document.getElementById('termsError');
                if (!termsInput.checked) {
                     termsError.classList.remove('hidden');
                     isValid = false;
                } else {
                     termsError.classList.add('hidden');
                }

                if (isValid) {
                    // Simulate API Call
                    const btn = form.querySelector('button[type="submit"]');
                    const originalText = btn.innerHTML;
                    btn.innerHTML = '<span class="material-symbols-outlined animate-spin mr-2">progress_activity</span> Creating...';
                    btn.disabled = true;
                    btn.classList.add('opacity-70', 'cursor-not-allowed');

                    setTimeout(() => {
                        alert('Account created successfully! (Simulation)');
                        btn.innerHTML = originalText;
                        btn.disabled = false;
                        btn.classList.remove('opacity-70', 'cursor-not-allowed');
                        form.reset();
                    }, 1500);
                }
            });
        });
    </script>
