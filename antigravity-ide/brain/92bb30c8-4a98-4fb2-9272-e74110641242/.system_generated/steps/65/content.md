<script>
        document.addEventListener('DOMContentLoaded', () => {
            const form = document.getElementById('loginForm');
            const emailInput = document.getElementById('email');
            const passwordInput = document.getElementById('password');
            const emailErrorMsg = document.getElementById('emailErrorMessage');
            const emailErrorIcon = document.getElementById('emailErrorIcon');
            const passwordErrorMsg = document.getElementById('passwordErrorMessage');
            
            const togglePasswordBtn = document.getElementById('togglePassword');
            const toggleIcon = document.getElementById('toggleIcon');

            // Toggle Password Visibility
            togglePasswordBtn.addEventListener('click', () => {
                const type = passwordInput.getAttribute('type') === 'password' ? 'text' : 'password';
                passwordInput.setAttribute('type', type);
                toggleIcon.textContent = type === 'password' ? 'visibility_off' : 'visibility';
            });

            // Simulate form submission and trigger error states as requested by the prompt
            form.addEventListener('submit', (e) => {
                e.preventDefault(); // Prevent actual submission

                // Reset states briefly for animation effect
                emailInput.classList.remove('border-error', 'focus:ring-error/20', 'focus:border-error');
                emailInput.classList.add('border-outline-variant', 'focus:ring-primary/20', 'focus:border-primary');
                passwordInput.classList.remove('border-error', 'focus:ring-error/20', 'focus:border-error');
                passwordInput.classList.add('border-outline-variant', 'focus:ring-primary/20', 'focus:border-primary');
                
                emailErrorMsg.classList.add('hidden');
                emailErrorIcon.classList.add('hidden');
                passwordErrorMsg.classList.add('hidden');

                // Simulate validation delay
                setTimeout(() => {
                    // Show Email Error
                    emailInput.classList.remove('border-outline-variant', 'focus:ring-primary/20', 'focus:border-primary');
                    emailInput.classList.add('border-error', 'focus:ring-error/20', 'focus:border-error');
                    emailErrorMsg.classList.remove('hidden');
                    emailErrorIcon.classList.remove('hidden');

                    // Show Password Error
                    passwordInput.classList.remove('border-outline-variant', 'focus:ring-primary/20', 'focus:border-primary');
                    passwordInput.classList.add('border-error', 'focus:ring-error/20', 'focus:border-error');
                    passwordErrorMsg.classList.remove('hidden');
                    
                    // Add shake animation class (defined via arbitrary tailwind or style tag if needed, keeping it simple here with just color states)
                }, 150);
            });
        });
    </script>
